{%- comment -%} Renders an edition's workshop date, or a range when `event_end_date` is set. With `event_date_tentative`, the two dates are shown as alternatives (A/B). Params: ed (edition), weekday (true to include day names). {%- endcomment -%} {%- assign _start = include.ed.event_date -%} {%- assign _end = include.ed.event_end_date -%} {%- if _start and _end -%} {%- assign _sm = _start | date: "%Y-%m" -%}{%- assign _em = _end | date: "%Y-%m" -%} {%- if include.ed.event_date_tentative -%}{%- assign _sep = "/" -%}{%- assign _wsep = " / " -%}{%- else -%}{%- assign _sep = "–" -%}{%- assign _wsep = " – " -%}{%- endif -%} {%- if include.weekday -%}{{ _start | date: "%A" }}{{ _sep }}{{ _end | date: "%A" }}, {% endif -%} {%- if _sm == _em -%} {{ _start | date: "%B %-d" }}{{ _sep }}{{ _end | date: "%-d, %Y" }} {%- else -%} {{ _start | date: "%B %-d" }}{{ _wsep }}{{ _end | date: "%B %-d, %Y" }} {%- endif -%} {%- elsif _start -%} {%- if include.weekday -%}{{ _start | date: "%A, %B %d, %Y" }}{%- else -%}{{ _start | date: "%B %d, %Y" }}{%- endif -%} {%- endif -%}
